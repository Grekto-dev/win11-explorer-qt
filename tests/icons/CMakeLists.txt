if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.21)
    project(explorericons-tests LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    enable_testing()
endif()
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core Gui Widgets Test)
set(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
file(GLOB_RECURSE ICON_FILES CONFIGURE_DEPENDS "${REPO_ROOT}/icons/explorer/*")
add_executable(tst_explorericons tst_explorericons.cpp ${REPO_ROOT}/pcmanfm/explorericons.cpp)
qt6_add_resources(tst_explorericons explorer_icons PREFIX "/icons/win11-explorer-qt" BASE "${REPO_ROOT}/icons/explorer" FILES ${ICON_FILES})
target_include_directories(tst_explorericons PRIVATE "${REPO_ROOT}/pcmanfm")
target_link_libraries(tst_explorericons PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME explorer-icons COMMAND tst_explorericons)
set_tests_properties(explorer-icons PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
